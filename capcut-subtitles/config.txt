# ===== 자막 설정 =====
# 줄 맨 앞에 # 이 붙은 줄은 설명입니다. "이름 = 값"의 값만 바꾸세요.

# 자막 한 줄 최대 글자 수 (띄어쓰기 포함)
#   세로 쇼츠/릴스: 14~20 추천,  가로 유튜브 영상: 25~35 추천
max_chars = 20

# 자막 하나가 화면에 떠 있는 최대 시간(초)
max_seconds = 6

# 문장 끝 마침표(.)와 쉼표(,) 지우기 (yes / no)
remove_period = yes

# 언어: ko = 한국어, en = 영어, ja = 일본어, auto = 자동 감지
language = ko

# 인식 모델
#   large-v3       = 가장 정확함 (기본값)
#   large-v3-turbo = 조금 덜 정확하지만 훨씬 빠름
model = large-v3

# 실행 장치: auto = GPU 있으면 GPU, 없으면 CPU / cpu = 항상 CPU
device = auto

# 싱크 보정: 받아쓴 글자를 음성에 다시 맞춰서 자막 시간을 정확하게 (yes / no)
#   처음 한 번 보정 모델(약 360MB)을 내려받고, 자막 만드는 시간이 조금 늘어나요
sync_fix = yes

# 자막에서 뺄 말버릇 (쉼표로 구분). "어?" "어!"처럼 물음표·느낌표가 붙은 건 남겨요
#   아무것도 빼지 않으려면 = 뒤를 비워 두세요
remove_words = 어, 어어, 음, 으음

# 말이 끝난 뒤 자막을 얼마나 더 보여줄지(초). 음수면 그만큼 일찍 내려요 (싱크 보정을 켰을 때)
#   자막이 너무 오래 남으면 더 작게 (예: -0.3), 너무 빨리 사라지면 더 크게 (예: 0.2)
end_pad = -0.1
